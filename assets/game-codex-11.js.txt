filter = e.target.value; render(); });

      els.dropdown.classList.remove("open");

      if (!confirm("最終確認：バックアップ済みですか？")) return;

      try {

        toast("全データを削除できませんでした。", "error");
      } finally {
        clearingAllData = false;
      }
    });

  async function init() {
    try {
      els.storageInfo.title = "IndexedDBに保存します。ブラウザーデータの削除やストレージ消去で失われる可能性があるため、重要なデータはバックアップ推奨です。";
      if (!("indexedDB" in window)) throw new Error("IndexedDB unsupported");
      await openDB();
      bindEvents();
      await refresh();
    } catch (err) {
      console.error(err);
      els.library.innerHTML = `<div class="empty"><span class="big">⚠️</span><b>保存機能を開始できませんでした</b><p>通常モードの最新版Chrome / Edge / Firefox / Safari等で開き、ブラウザーのサイトデータ保存が有効か確認してください。</p></div>`;
      toast("ブラウザのデータベース機能を利用できません。", "error");
    }
  }

  init();
