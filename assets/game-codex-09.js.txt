  function downloadCurrentHTML() {
    const guide = state.guides.find(g => g.id === state.viewingId);
    if (!guide || isUrlGuide(guide)) return;
    downloadBlob(guide.html, "text/html;charset=utf-8", `${safeFilename(guide.title)}.html`);
  }

  function openCurrentInNewTab() {
    const guide = state.guides.find(g => g.id === state.viewingId);
    if (!guide) return;
    if (isUrlGuide(guide)) {
      const link = document.createElement("a");
      link.href = guide.url;
      link.target = "_blank";
      link.rel = "noopener noreferrer";
      document.body.appendChild(link);
      link.click();
      link.remove();
      return;
    }
    const blob = new Blob([guide.html], {type:"text/html;charset=utf-8"});
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.target = "_blank";
    link.rel = "noopener noreferrer";
    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 60000);
  }

  async function handleCardAction(event) {
    const button = event.target.closest("[data-action]");
    if (!button) return;
    const card = button.closest(".card");
    const guideId = card?.dataset.id;
    if (!guideId) return;
    const action = button.dataset.action;
    if (action === "view") openViewer(guideId);
    if (action === "edit") openEditor(guideId);
    if (action === "favorite") await toggleFavorite(guideId);
    if (action === "delete") {
      const guide = state.guides.find(g => g.id === guideId);
      if (!guide) return;
      const approved = confirm(`「${guide.title}」を削除しますか？\n\n${isUrlGuide(guide) ? "登録URL" : "HTML本体・保存済み進捗"}・ジャケット画像・メモが削除されます。`);
      if (!approved) return;
      const key = String(guideId);
      blockedProgressIds.add(key);
      try {
        await queueGuideWrite(guideId, () => deleteGuide(guideId));
        await waitForGuideWrites(guideId);
        if (state.editingId === guideId) state.editingId = null;
        if (state.viewingId === guideId) closeModal(els.viewerModal);
        await refresh();
        toast(`「${guide.title}」を削除しました。`);
      } catch (error) {
        console.error(error);
        toast("削除できませんでした。もう一度お試しください。", "error");
      } finally {
        blockedProgressIds.delete(key);
      }
    }
  }

  function safeImportedTimestamp(value, fallback) {
    const n = Number(value);
    return Number.isFinite(n) && n > 0 ? n : fallback;
  }

  function safeImportedId(value) {
    const text = typeof value === "string" || typeof value === "number" ? String(value).trim() : "";
    return text && text.length <= 200 ? text : id();
  }

  function safeImportedCover(value) {
    if (typeof value !== "string" || !value.startsWith("data:image/")) return null;
    return value.length <= 24_000_000 ? value : null;
  }
