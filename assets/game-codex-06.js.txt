n);
    let tracked;
    tracked = run.finally(() => {
      if (guideWriteQueues.get(key) === tracked) guideWriteQueues.delete(key);
    });
    guideWriteQueues.set(key, tracked);
    return tracked;
  }

  async function waitForGuideWrites(guideId) {
    const key = String(guideId);
    while (guideWriteQueues.has(key)) {
      const pending = guideWriteQueues.get(key);
      await pending.catch(() => {});
      if (guideWriteQueues.get(key) === pending) break;
    }
  }

  async function waitForAllGuideWrites() {
    while (guideWriteQueues.size) {
      await Promise.allSettled([...guideWriteQueues.values()]);
    }
  }

    });
  }
