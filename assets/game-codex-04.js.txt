        ? `<img src="${escapeHTML(g.cover)}" alt="${escapeHTML(g.title)}のジャケット" loading="lazy">`

          ${cover}

          </div>
        </div>
        <div class="card-body">
          <div>
            <h3 title="${escapeHTML(g.title)}">${escapeHTML(g.title)}</h3>
            <div class="meta"><span>${escapeHTML(metaLabel)}</span><span>・</span><span>${formatDate(g.updatedAt)}</span></div>

          </div>
        </div>
      </article>`;
    }).join("");
  }

  async function addFiles(files) {
    const htmlFiles = [...files].filter(f => /\.html?$/i.test(f.name) || f.type === "text/html");
    if (!htmlFiles.length) return toast("HTMLファイルを選択してください。", "error");
    let added = 0;
    let lastAddedId = null;
    for (const file of htmlFiles) {
      try {
        const html = await fileToText(file);
        if (!html.trim()) throw new Error("empty");
        const now = Date.now() + added;
        const guide = {
          id: id(),
          sourceType: "html",
          url: "",
          title: parseTitle(html, file.name.replace(/\.html?$/i, "")),
          platform: "",
          tags: [],
          notes: "",
          favorite: false,
          cover: null,
          progressStorage: {},
          checkboxProgress: {},
          progressUpdatedAt: 0,
          html,
          originalName: file.name,
          createdAt: now,
          updatedAt: now
        };
        await putGuide(guide);
        lastAddedId = guide.id;
        added++;
      } catch (error) {
        console.error(error);
        toast(`${file.name} を読み込みまたは保存できませんでした。`, "error");
      }
    }
    els.htmlInput.value = "";
    await refresh();
    if (added > 0) toast(`${added}件の攻略チャートを追加しました。`);
    if (added === 1 && lastAddedId) openEditor(lastAddedId);
  }

    els.editCoverPreview.src = dataUrl || "";

  }

    state.editingId = guideId;
    state.pendingCover = guide.cover || null;
    state.pendingCoverChanged = false;
    els.titleInput.value = guide.title || "";
    els.platformInput.value = guide.platform || "";
    const web = isUrlGuide(guide);
    els.editHeading.textContent = web ? "攻略サイトを編集" : "攻略HTMLを編集";
